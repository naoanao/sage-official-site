      object: 'block', type: 'paragraph',
      paragraph: { rich_text: [{ text: { content: content.slice(0, 2000) } }] }
    }]
  };

  try {
    // ── server.py プロキシ経由で保存 ──────────────────
    const res  = await fetch(notionProxyUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });
    const data = await res.json();
    if (res.ok && data.id) {
      resultEl.textContent = '✅ Notionに保存しました！';
      resultEl.className   = 'nm-result ok';
      toast('📓 Notionに保存しました ✓', 'ok');
      setTimeout(() => closeNotionModal(), 1800);
    } else {
      throw new Error(data.message || data.error || `HTTP ${res.status}`);
    }
  } catch(e) {
    resultEl.textContent = `❌ 保存に失敗：${e.message.slice(0, 80)}`;
    resultEl.className   = 'nm-result err';
    toast('Notion保存に失敗しました', 'err');
  }
  btn.disabled = false; btn.textContent = '📓 Notionに保存';
}
